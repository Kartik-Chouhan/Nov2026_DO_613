--> Ansible - Deploy lambda functions using AWS CLI

- name: Deploy AWS lambda function
  hosts: localhost
  connection: local    #This is added, it means don't connect to localhost over SSH.
  gather_facts: false  #This is added, actually by default ansible collects the facts(i.e. Information) from managed hosts like hostname, IP add, memory etc. so false means it will not collect those infos.
					   #It is false here because, we are running on localhost, no need to collect any info/facts.
  tasks:
	- name: Create lambda function
	  ansible.builtin.command: 
	  cmd: > aws lambda create-function 
			 --function-name day23-demo-lambda 
		 	 --runtime python3.12 # Here, using 3.12 as the updated version
			 --role arn:aws:iam::123456789012:role/lambda-execution-role 
			 --handler lambda_function.lambda_handler 
			 --zip-file fileb://lambda.zip
	 
	  register: lambda_result  #Register 	is to capture the output means it stores the output/result of a task in a variable.
	  changed_when: true       #It explicitly tells Ansible to mark the task as changed whenever it succeeds
--> Terraform -- Serverless API gateway with lambda integration

resource "aws_api_gateway_rest_api" "demo_api" {
	name 			= "day23_demo_api"
	description 	= "Demo API for learning purpose for Day23"
}

resource "aws_lambda_function" "day23_demo_lambda" {
	function_name 		= "day23-demo-lambda"
	runtime 			= "python3.12"
	role 				= aws_iam_role.lambda_exec_role.arn
	handler				= "lambda_function.lambda_handler"
	filename 			= "lambda.zip"
	source_code_hash = filebase64sha256("lambda.zip")
}

resource "aws_api_gateway_resource" "demo_resource" {
	rest_api_id 	= aws_api_gateway_rest_api.demo_api.id
	parent_id 		= aws_api_gateway_rest_api.demo_api.root_resource_id
	patch_part 		= "hello"
}

resource "aws_api_gateway_method" "demo_method" {

	rest_api_id 	= aws_api_gateway_rest_api.demo_api.id
	resource_id		= aws_api_gateway_resource.demo_resource.id
	http_method		= "GET"
	authorization	= "NONE"	
}

resource "aws_api_gateway_integration" "lambda_integration" {
	rest_api_id   	= aws_api_gateway_rest_api.demo_api.id
	resource_id		= aws_api_gateway_resource.demo_resource.id
	http_method 	= aws_api_gateway_method.demo_method.http_method
	integration_http_method		= "POST"
	type 						= "AWS_PROXY"
	uri							= aws_lambda_function.demo_lambda.invoke_arn
}

--> Jenkinsfile -- Deploy Serverless Apps with lambda

pipeline {
	agent any
	stages {
		stage('Build') {
			steps{
				echo "Builduing Lambda Function..."
				script {				
					sh 'zip -r lambda.zip lambda/'			
				}
			}
		}
		stage('Deploy to AWS Lambda') {
			steps {
				script {
					sh 'aws lambda update-function-code --function-name day23-demo-lambda --zip-file fileb://lambda.zip'
				}				
			}
		}
	}
}

--> Docker
# Multi-Stage Docker Build

#Stage-1: Build stage

FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/usr/java-build
COPY . /opt/usr/java-build/
RUN mvn clean package

#Stage-2: Production Stage

FROM tomcat:9
WORKDIR /usr/local/tomcat/webapps
COPY --from=builder /opt/usr/java-build/target/*.war /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh", "run"]
