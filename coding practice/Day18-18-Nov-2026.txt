--> Ansible -- Configure HA Proxy for load balancing

```yaml
- name: Setup HA proxy for load balancing      #HAProxy distributes incoming traffic across multiple backend servers to avoid overloading a single server.
  hosts: all
  become: yes
  
  tasks:
    - name: Install HAProxy
	  yum:
		name: haproxy
		state: present

	- name: Configure HAProxy 
	  template:
		src: haproxy.cfg.j2  #haproxy.cfg.j2 must define the frontend, backend servers, and balancing algorithm.
		dest: /etc/haproxy/haproxy.cfg 
	  notify: Restart HAProxy
		
	- name: Start HAProxy service
	  service: 
		name: haproxy
		state: started
		enabled: yes
		
  handlers:
	- name: Restart HAProxy
	  service:
		name: haproxy
		state: restarted
		
	  
--> Terraform -- Multi-Region Auto scaling group with ALB

	name 			   = "day18_demo_lb"

	load_balancer_type = "application"
	security_groups = [aws_security_group.allow_ssh_http.id]

		aws_subnet.public.id,

	]

	name  				= "day18_demo_asg"

		aws_subnet.public.id,

	]
	launch_template = {
		id 		= aws_launch_configuratio.demo_config.id
		version = "$Latest"
	}
}

--> K8s -- Horizontal Pod Autoscaler with multiple replicas

apiVersion: apps/v1
kind: Deployment
metadata:
	name: demo-web-app
spec:
	replicas: 3
	selector:
		matchLabels:
			app: demo-web-app
	template:
		metadata:
			labels:
				app: demo-web-app
		spec:
			containers:
				- name: nginx
				  image: nginx
				  ports:	
					- containerPort: 80
				  
---

apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
	name: demo-web-app-hpa
spec:
	scaleTargetRef:
		apiVersion: apps/v1
		kind: Deployment
		name: demo-web-app
	minReplicas: 2
	maxReplicas: 10
	
	targetCPUUtilizationPercentage: 80  ## This is deprecated one with v2 so using below
	
	metrics:
		- type: Resource
		  resource:
			 name: cpu
			 target: 
				type: Utilization
				averageUtilization: 80

		
--> Jenkinsfile -- Blue green deployment strategy

pipeline {
	agent any 
	stages {
		stage('Build') {
			steps{
				echo "Building docker image...."
				docker.build("demo-web-app")
			}
		}
		stage('Test') {
			steps {
				echo "Testing cases...."
			}
		}
		stage('Blue Green Deployment') {
			steps {
				script {
					sh 'kubectl apply -f kubernetes/blue-deployment.yaml'
					sh 'kubectl apply -f kubernetes/green-deployment.yaml'
					sh 'kubectl rollout status deployment/blue-deployment'
					sh 'kubectl rollout status deployment/green-deployment' 
				}				
			}
		}
	}
}

--> Gitlab CI/CD -- Blue-Green deployment with K8s

stages:
	- build
	- push
	- deploy
	
build:
	stage: build
	script: 
		- docker build -t demo-web-app .
		
push:
	stage: push
	script: 
		- docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASS"
		- docker push demo-web-app
		
deploy:
	stage: deploy
	script:
		 - kubectl apply -f kubernetes/blue-deployment.yaml
		 - kubectl apply -f kubernetes/green-deployment.yaml
		 - kubectl rollout status deployment/blue-deployment
		 - kubectl rollout status deployment/green-deployment
