--> Ansible -- Auto healing with servers with Systemd

```yaml
- name: Configure auto healing for web servers
  hosts: all
  become: yes
  
  tasks:
    - name: Ensure Nginx service is running
	  service:
		- name: day17-nginx-service
		  state: started
		  enabled: yes		  
		  
	- name: Setup auto-restart for nginx service 
	  lineinfile:
		path: /etc/systmd/system/nginx.service.d/override.conf
		line: "Restart=always"
		create: yes
	
	- name: Reload systemd
	  command: systemctl daemon-reload

--> K8s -- Self-healing with liveness and readiness probes

apiVersion: app/v1
kind: Deployment
metadata:
	name: demo-web-app
spec:
	replicas: 3
	selector:
		matchLabels:
			app: demo-web-app
	template:
		metadata:
			labels:
				app: demo-web-app
		spec:
			containers:
				- name: nginx
				  image: nginx
				  ports:	
					- containerPort: 80
				  livenessProbe:
					httpGet:
						path: /healthz
						port: 80
					initialDelaySeconds: 5
					periodSeconds: 10
				  readinessProbe:
					httpGet:
						path: /healthz
						port: 80
					initialDelaySeconds: 5
					periodSeconds: 10
					
--> Jenkinsfile -- Rolling deployments with K8s

pipeline {
	agent any {
	environment { 
		IMAGE_NAME = 'demo-web-app'
	}
	stages {
		stage('Build') {
			steps{
				echo "Building docker image...."
				docker.build("${env.IMAGE_NAME}")
			}
		}
		stage('Test') {
			steps {
				echo "Testing cases...."
			}
		}
		stage('Deploy to K8s') {
			steps {
				script {
					sh 'kubectl apply -f kubernetes/deployment.yaml'
					sh 'kubectl rollout status deployment/demo-web-app' 
				}				
			}
		}
	}
}

--> Gitlab CI/CD -- Multi-cloud deployment

stages:
	- build
	- push
	- deploy
	
build:
	stage: build
	script: 
		- docker build -t demo-web-app .
		
push:
	stage: push
	script: 
		- docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASS"
		- docker push demo-web-app
		
deploy_aws:
	stage: deploy
	script:
	  	- kubectl --context=aws-cluster apply -f kubernetes/deployment.yaml

deploy_azure:
	stage: deploy
	script: 
		- kubectl --context=aws-cluster apply -f kubernetes/deployment.yaml
