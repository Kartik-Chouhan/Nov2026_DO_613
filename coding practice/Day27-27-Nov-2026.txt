Day 27 -- Serverless scaling with Lambda and DynamoDB

--> Ansible - Deploy serverless Lambda function

- name: Deploy serverless lambda function
  hosts: localhost
  become: yes
  gather_facts: false
  tasks:
	- name: Create Lambda Function
		command: aws lambda create-function 
				 --function-name demo-lambda 
				 --runtime nodejs22/x 
				 --role arn:aws:iam::12345678912:role/lambda-role 
				 --handler index.handler       # Here, handler is the entry point of the lambda function
				 --zip-file fileb://lambda.zip  # Here, fileb:// uploads the zip file as binary data
     
--> Terraform -- Create DynamoDB table for serverless application

resource "aws_dynamodb_table" "demo_table" {
	name 			= "demo-table"
	hash_key 		= "id"    # Here, hash_key defines the partition key
	billing_mode	= "PAY_PER_REQUEST"
	attribute {
		name = "id"
		type = "S"   # Here, S defines the string attribute
	}
}

--> K8s -- Deploy Serverless using Kubeless

apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
	name: demo-lambda
	namespace: default
spec:
	runtime:	nodejs22
	handler: index.handler
	function: |
		module.exports = function(event, context) {
			context.succeed('Hello World');
		};
	
--> Jenkinsfile -- Lambda Function Deployment Pipeline
pipeline {
	agent any
	stages {
		stage('Build') {
			steps {
				echo 'Building lambda function package...'
				sh 'zip -r lambda.zip lambda/'
			}
		}
		stage('Deploy to Lambda') {
			steps {
				script {
					sh 'aws lambda update-function-code 
					   --function-name demo-lambda 
					   --zip-file fileb://lambda.zip'
				}
			}
		}
	}
}
