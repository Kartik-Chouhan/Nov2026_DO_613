--> Ansible: - Multi-node Nginx setup

  hosts: all
  become: yes
  tasks:
    - name: Install nginx
	  yum:

		state: present
	
	- name:start nginx service
	  service: 
		name: nginx
		state: started

--> Terraform - create IAM users and policies

resource "aws_iam_user" "demo_user" {
  name = "policies_for_demo_task"
}

resource "aws_iam_policy" "demo_policy" {
  name        = "demo-s3-list-policy"
  description = "Policy for demo application"

  policy = jsonencode({   ## To convert the terraform config into JSON
    Version = "2012-10-17"

    Statement = [
      {
        Effect   = "Allow"
        Action   = "s3:ListBucket"
        Resource = "arn:aws:s3:::pathnex-bucket"
      }
    ]
  })
}

resource "aws_iam_user_policy_attachment" "demo_attachment" {
  user       = aws_iam_user.demo_user.name
  policy_arn = aws_iam_policy.demo_policy.arn
}

--> K8s -- deploy Nginx with configMap and secrets

apiVersion: v1
kind: ConfigMap  ## ConfigMap stores non-sensitive configuration
metadata:
	name: nginx-config
data:
	nginx.conf: |
		server {
			listen 80;
			server_name localhost;
			location {
				root /usr/share/nginx/html;
				index index.html index.htm;
			}
		}
	
---
apiVersion: v1
kind: Secret ## Secret stores sensitive information.
metadata:
	name: nginx-secret
type: Opaque
stringData: 
	password: my_password

---
apiVersion: apps/v1
kind: Deployment
metadata: 
	name: kartik-nginx
spec:
	replicas: 2
	selector:
		matchLabels:
			app: kartik-nginx
		template:
			metadata:
				labels:
					app: kartik-nginx
			spec:
				containers:
					- name: nginx
					  image: nginx
					  volumeMounts:
						- mountPath: /etc/nginx/nginx.conf
						  name: nginx-config
						  subPath: nginx.conf
				volumes:
					- name: nginx-config
					  configMap:
						name: nginx-config
					- name: nginx-secret
						secret:
							secretName: nginx-secret

--> Jenkinsfile -- Multi-branch pipeline with docker build

pipeline {
	agent any 
	stages {
		stage('Build') {
			steps {
				script {
					docker.build('demo-app:latest')
				}
			}
		}
		stage('Deploy') {
			steps {
				script {
					sh kubectl apply -f kubernetes/deploy.yaml
				}
			}
		}
	}
}

--> Gitlab CI/CD -- Advanced Deployment with Helm

stages:
	- build
	- push
	- deploy
	
variables:
  IMAGE_NAME: "$CI_REGISTRY_IMAGE:$CI_COMMIT_SHORT_SHA"
	
build: 
	stage: build
	script:
	  - docker build -t "$IMAGE_NAME" .
	
	push: 
	  stage: push
	  script:
		- docker login -u "$REGISTRY_USER" -p "$REGISTRY_PASS"
		- docker push demo-web-app
		
	deploy:
		stage: deploy
		script:
		  - helm repo add pathnex https://charts.pathnex.com
		  - helm repo update
		  - helm upgrad --install pathnex-nginx pathnex/nginx-ingress
