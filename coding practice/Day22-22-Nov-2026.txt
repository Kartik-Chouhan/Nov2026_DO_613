Day 22 -- Multi- cluster K8s management

--> Ansible - Deploy Nginx across multiple K8s cluster

- name: Deploy Nginx across multiple K8s cluster
  hosts: localhost
  become: yes
  
  tasks:
	- name: Configure kubectl for cluster 1
	  ansible.builtin.command
	  cmd: kubectl --context=us-east config use-context us-east-cluster
	  
	- name: Deploy Nginx on cluster 1
	  cmd: kubectl apply -f kubernetes/deployment.yaml
	  
	- name: Configure kubectl for cluster 2
	  ansible.builtin.command
	  cmd: kubectl --context=us-west config use-context us-west-cluster
	  
	- name: Deploy Nginx on cluster 2
	  ansible.builtin.command
	  cmd: kubectl apply -f kubernetes/deployment.yaml

--> K8s -- Federated Deployment with Multiple Clusters

## Federation means coordinating resources and policies across multiple clusters through a federation control plane.

apiVersion: apps/v1
kind: Deployment
metadata:
	name: demo-deployment
spec:
	replicas: 3
	selector:
		matchLabels:
			app: demo-app
	template:
		metadata:
			labels:
				app: demo-app
	spec:
		containers:
			- name: nginx
			  image: nginx
			  ports:
				- containerPort: 80

---
apiVersion: federation.k8s.io/v1beta1
kind: FederratedDeployment
metadata:

	name: demo-federated-deployment
	namespace: default
spec:
	template:
		spec:
			replicas: 3
			selector:
				matchLabels:
					app: demo-app
			template:
				metadata:
					labels:
						app: demo-app
				spec:
					containers:
						- name: nginx
						  image: nginx
						  ports:
							- containerPort: 80
					
--> Jenkinsfile -- Multi-Cluster Deployment Strategy

pipeline {
	agent any
	stages {
		stage('Deploy to Cluster 1') {
			steps{
				script {				
					sh 'kubectl --context=us-east-cluster apply -f kubernetes/deployment.yaml'			
				}
			}
		}
		stage('Deploy to Cluster 2') {
			steps {
				script {
					sh 'kubectl --context=us-west-cluster apply -f kubernetes/deployment.yaml'
				}				
			}
		}
	}
}
